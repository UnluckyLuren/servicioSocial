<script>
/**
 * ============================================================
 * CUCEI COWORKING SYSTEM – Controlador Frontend v2.0
 * script.js
 * ============================================================
 */

(function (global) {
  'use strict';

  // ──────────────────────────────────────────────────────────
  // MÓDULO: GasApi
  // ──────────────────────────────────────────────────────────
  const GasApi = {
    ejecutar(nombreFuncion, ...args) {
      return new Promise((resolve, reject) => {
        if (typeof google === 'undefined' || !google.script) {
          reject(new Error('google.script.run no está disponible.'));
          return;
        }
        google.script.run
          .withSuccessHandler(resolve)
          .withFailureHandler(reject)
          [nombreFuncion](...args);
      });
    }
  };

  // ──────────────────────────────────────────────────────────
  // MÓDULO: AuthService
  // ──────────────────────────────────────────────────────────
  const AuthService = {
    usuario: null,

    async inicializar() {
      try {
        const sesionGuardada = localStorage.getItem('coworking_sesion');
        if (sesionGuardada) {
          this.usuario = JSON.parse(sesionGuardada);
          this._renderizarVista(this.usuario);
        } else {
          this.usuario = null;
          this._renderizarVistaPublica();
        }
      } catch (error) {
        console.error('[AuthService] Error al inicializar:', error);
        this._renderizarVistaPublica();
      } finally {
        const pantallaGlobal = document.getElementById('pantallaGlobalCarga');
        if (pantallaGlobal) pantallaGlobal.classList.add('hidden');
      }
    },

    _renderizarVistaPublica() {
      document.getElementById('headerPrincipal')?.classList.remove('hidden');
      document.getElementById('headerAuthButtons')?.classList.remove('hidden');
      document.getElementById('headerUsuarioLogueado')?.classList.add('hidden');
      document.getElementById('vistaAdmin')?.classList.add('hidden');
      document.getElementById('vistaAlumno')?.classList.remove('hidden');
    },

    _renderizarVista(info) {
      document.getElementById('headerPrincipal')?.classList.remove('hidden');

      if (!info.accesoConcedido) {
        this._mostrarVistaDenegada(info.mensaje || 'Acceso no autorizado.');
        return;
      }

      document.getElementById('headerAuthButtons')?.classList.add('hidden');
      const headerUser = document.getElementById('headerUsuarioLogueado');
      if (headerUser) {
        headerUser.classList.remove('hidden');
        headerUser.classList.add('flex');
      }

      this._actualizarHeader(info);

      if (info.rol === 'admin') {
        document.getElementById('vistaAlumno')?.classList.add('hidden');
        document.getElementById('vistaAdmin')?.classList.remove('hidden');
        document.getElementById('btnMisReservas')?.classList.add('hidden');
        AdminDashboard.inicializar();
      } else {
        document.getElementById('vistaAdmin')?.classList.add('hidden');
        document.getElementById('vistaAlumno')?.classList.remove('hidden');
        // Mostrar botón Mis Reservas
        const btnMis = document.getElementById('btnMisReservas');
        if (btnMis) {
          btnMis.classList.remove('hidden');
          btnMis.classList.add('flex');
        }
        StudentView.inicializar(info);
      }
    },

    _actualizarHeader(info) {
      const correoWrapper = document.getElementById('headerCorreoUsuario');
      const correoTexto   = document.getElementById('textoCorreoHeader');
      const badgeRol      = document.getElementById('headerBadgeRol');

      if (info.correo && correoTexto) {
        correoTexto.textContent = info.correo;
        correoWrapper?.classList.remove('hidden');
        correoWrapper?.classList.add('flex');
      }

      if (badgeRol) {
        badgeRol.classList.remove('hidden');
        const rolClases = {
          admin:   'text-xs font-bold px-2.5 py-1 rounded-full bg-purple-500/20 text-purple-200 border border-purple-500/30',
          maestro: 'text-xs font-bold px-2.5 py-1 rounded-full bg-blue-500/20 text-blue-200 border border-blue-500/30',
          alumno:  'text-xs font-bold px-2.5 py-1 rounded-full bg-emerald-500/20 text-emerald-200 border border-emerald-500/30'
        };
        badgeRol.className = rolClases[info.rol] || rolClases.alumno;
        badgeRol.textContent = info.rol.charAt(0).toUpperCase() + info.rol.slice(1);
      }
    },

    cerrarSesion() {
      localStorage.removeItem('coworking_sesion');
      this.usuario = null;
      location.reload();
    },

    _mostrarVistaDenegada(mensaje) {
      const msgEl = document.getElementById('mensajeDenegado');
      if (msgEl) msgEl.textContent = mensaje;
      document.getElementById('vistaDenegada')?.classList.remove('hidden');
      document.getElementById('vistaAlumno')?.classList.add('hidden');
      document.getElementById('vistaAdmin')?.classList.add('hidden');
      document.getElementById('headerAuthButtons')?.classList.remove('hidden');
      document.getElementById('headerUsuarioLogueado')?.classList.add('hidden');
    },

    abrirModalAuth(modo = 'login') {
      this.setAuthMode(modo);
      document.getElementById('modalAuth')?.classList.remove('hidden');
      document.body.style.overflow = 'hidden';
      this._ocultarMensajeAuth();
    },

    cerrarModalAuth() {
      document.getElementById('modalAuth')?.classList.add('hidden');
      document.body.style.overflow = 'auto';
      document.getElementById('formLogin')?.reset();
      document.getElementById('formRegister')?.reset();
    },

    setAuthMode(modo) {
      const formLogin = document.getElementById('formLogin');
      const formReg   = document.getElementById('formRegister');
      const btnLogin  = document.getElementById('btnTabLogin');
      const btnReg    = document.getElementById('btnTabRegister');

      this._ocultarMensajeAuth();

      if (modo === 'login') {
        formLogin?.classList.remove('hidden');
        formReg?.classList.add('hidden');
        if (btnLogin) btnLogin.className = 'flex-1 text-sm font-bold py-2 rounded-md bg-white shadow text-[#002f5c]';
        if (btnReg)   btnReg.className   = 'flex-1 text-sm font-bold py-2 rounded-md text-slate-500 hover:text-slate-700';
      } else {
        formLogin?.classList.add('hidden');
        formReg?.classList.remove('hidden');
        if (btnReg)   btnReg.className   = 'flex-1 text-sm font-bold py-2 rounded-md bg-white shadow text-[#002f5c]';
        if (btnLogin) btnLogin.className = 'flex-1 text-sm font-bold py-2 rounded-md text-slate-500 hover:text-slate-700';
      }
    },

    async procesarLogin(evento) {
      evento.preventDefault();
      const id   = document.getElementById('loginId').value.trim();
      const pass = document.getElementById('loginPass').value.trim();
      const btn  = document.getElementById('btnLoginSubmit');

      this._mostrarCargandoAuth(btn, true, 'Iniciando...');
      this._ocultarMensajeAuth();

      try {
        const res = await GasApi.ejecutar('loginUsuarioCustom', id, pass);
        if (res.exito) {
          localStorage.setItem('coworking_sesion', JSON.stringify(res.usuario));
          this.cerrarModalAuth();
          this.inicializar();
        } else {
          this._mostrarMensajeAuth(res.mensaje, 'error');
        }
      } catch (err) {
        this._mostrarMensajeAuth('Error de conexión.', 'error');
      } finally {
        this._mostrarCargandoAuth(btn, false, 'Iniciar Sesión');
      }
    },

    async procesarRegistro(evento) {
      evento.preventDefault();
      const correo = document.getElementById('regEmail').value.trim();
      const codigo = document.getElementById('regCodigo').value.trim();
      const pass   = document.getElementById('regPass').value.trim();
      const btn    = document.getElementById('btnRegSubmit');

      if (pass.length < 8) {
        this._mostrarMensajeAuth('La contraseña debe tener al menos 8 caracteres.', 'error');
        return;
      }

      this._mostrarCargandoAuth(btn, true, 'Registrando...');
      this._ocultarMensajeAuth();

      try {
        const res = await GasApi.ejecutar('registrarUsuarioCustom', correo, codigo, pass);
        if (res.exito) {
          this._mostrarMensajeAuth('Registro exitoso. Iniciando sesión...', 'exito');
          const loginRes = await GasApi.ejecutar('loginUsuarioCustom', correo, pass);
          if (loginRes.exito) {
            localStorage.setItem('coworking_sesion', JSON.stringify(loginRes.usuario));
            setTimeout(() => { this.cerrarModalAuth(); this.inicializar(); }, 1500);
          }
        } else {
          this._mostrarMensajeAuth(res.mensaje, 'error');
        }
      } catch (err) {
        this._mostrarMensajeAuth('Error de conexión.', 'error');
      } finally {
        this._mostrarCargandoAuth(btn, false, 'Crear Cuenta');
      }
    },

    _mostrarCargandoAuth(btn, activo, texto) {
      if (!btn) return;
      btn.disabled = activo;
      btn.innerHTML = activo
        ? `<i class="fa-solid fa-spinner fa-spin mr-2"></i>${texto}`
        : texto;
    },

    _mostrarMensajeAuth(texto, tipo) {
      const div = document.getElementById('formAuthMsg');
      if (!div) return;
      div.textContent = texto;
      div.className = `mb-4 p-3 rounded-lg text-sm font-semibold text-center ${
        tipo === 'error'
          ? 'bg-rose-50 text-rose-600 border border-rose-200'
          : 'bg-emerald-50 text-emerald-600 border border-emerald-200'
      }`;
      div.classList.remove('hidden');
    },

    _ocultarMensajeAuth() {
      document.getElementById('formAuthMsg')?.classList.add('hidden');
    }
  };

  // ──────────────────────────────────────────────────────────
  // MÓDULO: StudentView
  // ──────────────────────────────────────────────────────────
  const StudentView = {
    fechasSeleccionadas: [],   // [{ fecha: 'yyyy-mm-dd', horaEntrada: '08:00', horaSalida: '10:00' }]
    archivosSeleccionados: [], // [{ nombre, tipo, base64, tamano }]
    correoUsuario: '',
    codigoUsuario: '',
    usarMismoHorario: true,    // por defecto checkbox marcado

    HORAS: [
      '08:00','08:30','09:00','09:30','10:00','10:30',
      '11:00','11:30','12:00','12:30','13:00','13:30',
      '14:00','14:30','15:00','15:30','16:00','16:30',
      '17:00','17:30','18:00','18:30','19:00','19:30','20:00'
    ],

    inicializar(infoUsuario) {
      this.correoUsuario = infoUsuario.correo || '';
      this.codigoUsuario = infoUsuario.codigo || '';
      this._poblarSelectoresGlobales();
      // Cargar reservas devueltas en background
      this._cargarReservasDevueltas();
    },

    _poblarSelectoresGlobales() {
      const selectEntrada = document.getElementById('horaEntrada');
      if (!selectEntrada) return;
      selectEntrada.innerHTML = this.HORAS.slice(0, -1)
        .map(h => `<option value="${h}">${this._formatearAmPm(h)}</option>`)
        .join('');
      this.filtrarHoraSalida();
    },

    _formatearAmPm(horaStr) {
      const [h, m] = horaStr.split(':').map(Number);
      const ampm   = h >= 12 ? 'PM' : 'AM';
      const hora12 = h % 12 || 12;
      return `${String(hora12).padStart(2,'0')}:${String(m).padStart(2,'0')} ${ampm}`;
    },

    // ── Toggle: mismo horario para todas las fechas ──
    toggleMismoHorario() {
      const checkbox = document.getElementById('checkMismoHorario');
      this.usarMismoHorario = checkbox.checked;

      const horarioGlobal     = document.getElementById('horarioGlobal');
      const horariosPorFecha  = document.getElementById('horariosPorFecha');

      if (this.usarMismoHorario) {
        horarioGlobal?.classList.remove('hidden');
        horariosPorFecha?.classList.add('hidden');
      } else {
        horarioGlobal?.classList.add('hidden');
        horariosPorFecha?.classList.remove('hidden');
        this._renderizarHorariosPorFecha();
      }
    },

    _renderizarHorariosPorFecha() {
      const contenedor = document.getElementById('horariosPorFecha');
      if (!contenedor) return;
      contenedor.innerHTML = '';

      if (this.fechasSeleccionadas.length === 0) {
        contenedor.innerHTML = `
          <p class="text-xs text-slate-400 text-center py-2">
            <i class="fa-solid fa-circle-info mr-1"></i>Agrega fechas primero para configurar sus horarios individuales.
          </p>`;
        return;
      }

      this.fechasSeleccionadas.forEach((item, idx) => {
        const [yyyy, mm, dd] = item.fecha.split('-');
        const legible = `${dd}/${mm}/${yyyy}`;

        const div = document.createElement('div');
        div.className = 'bg-white border border-slate-200 rounded-xl p-3 animate-fadeIn';
        div.innerHTML = `
          <p class="text-xs font-black text-[#002f5c] mb-2 flex items-center gap-1.5">
            <i class="fa-solid fa-calendar-day"></i> ${legible}
          </p>
          <div class="grid grid-cols-2 gap-3">
            <div>
              <label class="block text-xs font-semibold text-slate-600 mb-1">Hora de Entrada</label>
              <select id="entrada_${idx}" onchange="StudentView._actualizarHoraSalida(${idx})"
                class="w-full border border-slate-200 rounded-lg px-2 py-1.5 text-xs bg-slate-50 focus:outline-none focus:ring-2 focus:ring-[#002f5c]/30">
                ${this.HORAS.slice(0,-1).map(h => `<option value="${h}" ${h === (item.horaEntrada||'08:00') ? 'selected' : ''}>${this._formatearAmPm(h)}</option>`).join('')}
              </select>
            </div>
            <div>
              <label class="block text-xs font-semibold text-slate-600 mb-1">Hora de Salida</label>
              <select id="salida_${idx}"
                class="w-full border border-slate-200 rounded-lg px-2 py-1.5 text-xs bg-slate-50 focus:outline-none focus:ring-2 focus:ring-[#002f5c]/30">
                ${this._opcionesSalidaParaEntrada(item.horaEntrada || '08:00', item.horaSalida)}
              </select>
            </div>
          </div>
        `;
        contenedor.appendChild(div);
      });
    },

    _opcionesSalidaParaEntrada(horaEntrada, horaSeleccionada) {
      const idx = this.HORAS.indexOf(horaEntrada);
      const opciones = this.HORAS.slice(idx + 1);
      return opciones.map(h => `<option value="${h}" ${h === horaSeleccionada ? 'selected' : ''}>${this._formatearAmPm(h)}</option>`).join('');
    },

    _actualizarHoraSalida(idx) {
      const entrada = document.getElementById(`entrada_${idx}`)?.value;
      const salida  = document.getElementById(`salida_${idx}`);
      if (!entrada || !salida) return;
      salida.innerHTML = this._opcionesSalidaParaEntrada(entrada, '');
      // Actualizar en array
      if (this.fechasSeleccionadas[idx]) {
        this.fechasSeleccionadas[idx].horaEntrada = entrada;
      }
    },

    seleccionarRol(rol) {
      document.getElementById('inputRol').value = rol;
      const badge        = document.getElementById('badgeRol');
      const labelCodigo  = document.getElementById('labelCodigo');
      const inputCodigo  = document.getElementById('codigo');
      const campoExt     = document.getElementById('campoExtension');
      const inputExt     = document.getElementById('extension');
      const campoResp    = document.getElementById('campoResponsable');

      if (this.correoUsuario) {
        const inputCorreo = document.getElementById('correo');
        if (inputCorreo) inputCorreo.value = this.correoUsuario;
      }
      if (this.codigoUsuario) {
        inputCodigo.value    = this.codigoUsuario;
        inputCodigo.readOnly = true;
        inputCodigo.className = 'w-full bg-slate-100 text-slate-500 border border-slate-200 rounded-lg px-3 py-2 text-sm cursor-not-allowed';
      }

      const hoy = new Date();
      const min = `${hoy.getFullYear()}-${String(hoy.getMonth()+1).padStart(2,'0')}-${String(hoy.getDate()).padStart(2,'0')}`;
      const selectorFecha = document.getElementById('selectorFecha');
      if (selectorFecha) selectorFecha.min = min;

    _procesarListaArchivos(fileList) {
      const TIPOS_PERMITIDOS = ['application/pdf','application/msword',
        'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
        'application/vnd.ms-excel',
        'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
        'image/png','image/jpeg'];
      const MAX_SIZE  = 5 * 1024 * 1024; // 5MB
      const MAX_FILES = 5;

      Array.from(fileList).forEach(file => {
        if (this.archivosSeleccionados.length >= MAX_FILES) {
          this._mostrarToast(`Máximo ${MAX_FILES} archivos permitidos.`, 'warn');
          return;
        }
        if (file.size > MAX_SIZE) {
          this._mostrarToast(`El archivo "${file.name}" supera los 5MB.`, 'error');
          return;
        }
        if (!TIPOS_PERMITIDOS.includes(file.type)) {
          this._mostrarToast(`Tipo de archivo no permitido: ${file.name}`, 'error');
          return;
        }
        if (this.archivosSeleccionados.some(a => a.nombre === file.name)) {
          return; // Duplicado
        }

        const reader = new FileReader();
        reader.onload = (e) => {
          // Extraer base64 sin el prefijo data:...;base64,
          const base64 = e.target.result.split(',')[1];
          this.archivosSeleccionados.push({
            nombre: file.name,
            tipo:   file.type,
            base64: base64,
            tamano: file.size
          });
          this._renderizarListaArchivos();
        };
        reader.readAsDataURL(file);
      });

      // Limpiar input para permitir re-selección
      document.getElementById('inputArchivos').value = '';
    },

    _renderizarListaArchivos() {
      const lista = document.getElementById('listaArchivos');
      if (!lista) return;
      lista.innerHTML = '';

      this.archivosSeleccionados.forEach((archivo, idx) => {
        const tamanoKb = (archivo.tamano / 1024).toFixed(1);
        const iconos = {
          'application/pdf': 'fa-file-pdf text-rose-500',
          'image/png':       'fa-file-image text-blue-500',
          'image/jpeg':      'fa-file-image text-blue-500'
        };
        const icono = iconos[archivo.tipo] || 'fa-file text-slate-500';

        const div = document.createElement('div');
        div.className = 'flex items-center justify-between bg-white border border-slate-200 rounded-lg px-3 py-2 animate-fadeIn';
        div.innerHTML = `
          <div class="flex items-center gap-2 min-w-0">
            <i class="fa-solid ${icono} flex-shrink-0"></i>
            <span class="text-xs font-semibold text-slate-700 truncate">${this._escapar(archivo.nombre)}</span>
            <span class="text-[10px] text-slate-400 flex-shrink-0">${tamanoKb} KB</span>
          </div>
          <button type="button" onclick="StudentView.eliminarArchivo(${idx})"
            class="text-slate-400 hover:text-rose-600 p-1 rounded hover:bg-rose-50 transition-colors flex-shrink-0 ml-2">
            <i class="fa-solid fa-trash-can text-xs"></i>
          </button>
        `;
        lista.appendChild(div);
      });
    },

    eliminarArchivo(idx) {
      this.archivosSeleccionados.splice(idx, 1);
      this._renderizarListaArchivos();
    },

      try {
        const res = await GasApi.ejecutar('obtenerDetalleReserva', idReserva, this.correoUsuario);
        if (!res.exito) {
          this._mostrarToast('No se pudo cargar la reserva.', 'error');
          return;
        }

        const r = res.reserva;
        // Buscar el coworking y abrir modal
        const botonesCoworking = document.querySelectorAll('[data-coworking]');
        let botonCoworking = null;

        botonesCoworking.forEach(btn => {
          if (btn.dataset.coworking === r.coworking) botonCoworking = btn;
        });

          // Abrir modal manualmente con datos básicos
          this._abrirModalConDatosDevueltos(r);
          return;
        }

        ModalController.abrir(botonCoworking);
        this._rellenarFormularioDevuelto(r);
        this.cerrarPanelMisReservas();

      } catch (e) {
        this._mostrarToast('Error al cargar la reserva: ' + e.message, 'error');
      }

        warn:  'bg-amber-50 text-amber-800 border border-amber-200'
      };
      el.className   = `mt-4 p-3 rounded-xl text-sm font-semibold text-center ${estilos[tipo] || estilos.error}`;
      el.textContent = texto;
      el.classList.remove('hidden');

        .replace(/"/g, '"')
        .replace(/'/g, ''');

    async inicializar() {
      try {
        const adminCorreo = AuthService.usuario?.correo || '';
        const datos = await GasApi.ejecutar('obtenerDashboardAdmin', adminCorreo);

        if (!datos.exito) {
          this._mostrarErrorDashboard(datos.mensaje);

    // ── Tabs ──
    cambiarTab(tab) {
      const tabs = ['reservas', 'calendario', 'usuarios'];
      const vistas = {
        reservas:   'adminVistaReservas',
        calendario: 'adminVistaCalendario',
        usuarios:   'adminVistaUsuarios'
      };
      const botones = {
        reservas:   'tabBtnReservas',
        calendario: 'tabBtnCalendario',
        usuarios:   'tabBtnUsuarios'
      };

      tabs.forEach(t => {
        const vistaEl = document.getElementById(vistas[t]);
        const btnEl   = document.getElementById(botones[t]);
        if (t === tab) {
          vistaEl?.classList.remove('hidden');
          if (btnEl) btnEl.className = 'px-4 py-2 text-sm font-bold border-b-2 border-[#002f5c] text-[#002f5c] whitespace-nowrap';
        } else {
          vistaEl?.classList.add('hidden');
          if (btnEl) btnEl.className = 'px-4 py-2 text-sm font-bold border-b-2 border-transparent text-slate-500 hover:text-slate-800 transition-colors whitespace-nowrap';
        }
      });

      // Inicializar calendario si se abre su pestaña
      if (tab === 'calendario') {
        setTimeout(() => this._inicializarCalendario(), 100);
      }

    // ── Calendario Digital con FullCalendar ──
    _inicializarCalendario() {
      const contenedor = document.getElementById('contenedorCalendario');
      if (!contenedor) return;

      // Destruir instancia previa si existe
      if (this._calendarioInstance) {
        this._calendarioInstance.destroy();
        this._calendarioInstance = null;
      }

      if (typeof FullCalendar === 'undefined') {
        contenedor.innerHTML = `

          FullCalendar no disponible. Verifica la conexión a internet.
        `;
        return;
      }

      const eventos = this._reservasAEventosCalendario(this.todasLasReservas);

      this._calendarioInstance = new FullCalendar.Calendar(contenedor, {
        initialView: 'dayGridMonth',
        locale: 'es',
        headerToolbar: {
          left:   'prev,next today',
          center: 'title',
          right:  'dayGridMonth,timeGridWeek,timeGridDay'
        },
        buttonText: {
          today:        'Hoy',
          month:        'Mes',
          week:         'Semana',
          day:          'Día'
        },
        events: eventos,
        eventClick: (info) => {
          const idReserva = info.event.extendedProps.idReserva;
          if (idReserva) {
            this.cambiarTab('reservas');
            setTimeout(() => this.abrirModal(idReserva), 300);
          }
        },
        eventDidMount: (info) => {
          // Tooltip básico con el nombre del evento
          info.el.title = `${info.event.title}\n${info.event.extendedProps.solicitante}\n${info.event.extendedProps.horario}`;
        },
        height: 'auto',
        eventTimeFormat: {
          hour:   '2-digit',
          minute: '2-digit',
          hour12: true
        }
      });

      this._calendarioInstance.render();
    },

    _reservasAEventosCalendario(reservas) {
      const colores = {
        'Pendiente': { bg: '#f59e0b', border: '#d97706', text: '#fff' },
        'Aceptada':  { bg: '#10b981', border: '#059669', text: '#fff' },
        'Rechazada': { bg: '#ef4444', border: '#dc2626', text: '#fff' },
        'Devuelta':  { bg: '#8b5cf6', border: '#7c3aed', text: '#fff' }
      };

      return reservas.map(r => {
        const color  = colores[r.estado] || { bg: '#94a3b8', border: '#64748b', text: '#fff' };
        const inicio = r.fecha && r.horaInicio ? `${r.fecha}T${r.horaInicio}` : r.fecha;
        const fin    = r.fecha && r.horaFin    ? `${r.fecha}T${r.horaFin}`    : r.fecha;

        return {
          id:              r.idReserva,
          title:           `${r.coworking.replace('Coworking ', 'CW')} · ${r.actividad || 'Sin nombre'}`,
          start:           inicio,
          end:             fin,
          backgroundColor: color.bg,
          borderColor:     color.border,
          textColor:       color.text,
          extendedProps: {
            idReserva:   r.idReserva,
            solicitante: r.nombreCompleto,
            horario:     `${r.horaInicio} – ${r.horaFin}`,
            estado:      r.estado,
            coworking:   r.coworking
          }
        };
      });
    },

    filtrarCalendario() {
      if (!this._calendarioInstance) return;

      const espacio = document.getElementById('filtroCalendarioEspacio')?.value || '';
      const estado  = document.getElementById('filtroCalendarioEstado')?.value  || '';

      let filtradas = [...this.todasLasReservas];
      if (espacio) filtradas = filtradas.filter(r => r.coworking === espacio);
      if (estado)  filtradas = filtradas.filter(r => r.estado    === estado);

      const eventos = this._reservasAEventosCalendario(filtradas);

      // Reemplazar eventos en el calendario
      this._calendarioInstance.removeAllEvents();
      eventos.forEach(e => this._calendarioInstance.addEvent(e));
    },

    async confirmarRechazo() {
      const motivo = document.getElementById('inputMotivoRechazo')?.value.trim();
      if (!motivo) {
        document.getElementById('inputMotivoRechazo')?.focus();
        document.getElementById('inputMotivoRechazo')?.classList.add('ring-2','ring-rose-400');
        setTimeout(() => document.getElementById('inputMotivoRechazo')?.classList.remove('ring-2','ring-rose-400'), 2000);
        return;
      }

      const permitirReenvio = document.getElementById('checkPermitirReenvio')?.checked || false;
      const nuevoEstado     = permitirReenvio ? 'Devuelta' : 'Rechazada';
      const idReserva       = this._reservaEnModal?.idReserva;

      if (!idReserva) return;

      // Mostrar cargando en el panel
      const panelRechazo = document.getElementById('panelRechazo');
      if (panelRechazo) {
        panelRechazo.innerHTML = `

             Procesando…
          `;
      }

      await this.cambiarEstado(idReserva, nuevoEstado, motivo);
    },

      document.body.style.overflow = 'auto';

    async cambiarEstado(idReserva, nuevoEstado, motivoRechazo = '') {
      const accionesEl  = document.getElementById('modalAdminAcciones');
      const adminCorreo = AuthService.usuario?.correo || '';

      if (accionesEl && !motivoRechazo) {
        accionesEl.innerHTML = `

             Procesando…
          `;
      }

      try {
        const resultado = await GasApi.ejecutar(
          'actualizarEstadoReserva',
          idReserva,
          nuevoEstado,
          adminCorreo,
          motivoRechazo
        );

        if (resultado.exito) {
          // Actualizar en memoria
          const idx = this.todasLasReservas.findIndex(r => r.idReserva === idReserva);
          if (idx !== -1) {
            this.todasLasReservas[idx].estado = nuevoEstado;
            if (motivoRechazo) this.todasLasReservas[idx].motivoRechazo = motivoRechazo;
          }

          // Actualizar badge del estado en el modal
          const estadoEl = document.getElementById('modalAdminEstado');
          if (estadoEl) {
            estadoEl.className   = `text-xs font-bold px-3 py-1 rounded-full ${this._obtenerClaseBadge(nuevoEstado)}`;
            estadoEl.textContent = nuevoEstado;
          }

          if (accionesEl) {
            accionesEl.classList.remove('hidden');
            accionesEl.innerHTML = `

                 ${resultado.mensaje}
              `;
          }

          // Ocultar panel de rechazo si estaba visible
          document.getElementById('panelRechazo')?.classList.add('hidden');

          setTimeout(() => {
            this.cerrarModal();
            this.filtrarPorEstado(this.filtroActivo);
          }, 2000);

        } else {
          if (accionesEl) {
            accionesEl.classList.remove('hidden');
            accionesEl.innerHTML = `

                 ${resultado.mensaje}
              `;
          }
          if (this._reservaEnModal) {
            setTimeout(() => {
              accionesEl?.classList.remove('hidden');
              this._renderizarBotonesModal(this._reservaEnModal);
              document.getElementById('panelRechazo')?.classList.add('hidden');
            }, 3000);
          }
        }
      } catch (error) {
        if (accionesEl) {
          accionesEl.classList.remove('hidden');
          accionesEl.innerHTML = `

              Error de conexión: ${error.message || 'Inténtalo de nuevo.'}
            `;
        }
        if (this._reservaEnModal) {
          setTimeout(() => {
            accionesEl?.classList.remove('hidden');
            this._renderizarBotonesModal(this._reservaEnModal);

function hashPassword_(password) {
  const rawHash = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256, password
  );
  let txtHash = '';
  for (let i = 0; i < rawHash.length; i++) {
    let v = rawHash[i];
    if (v < 0) v += 256;
    txtHash += (v < 16 ? '0' : '') + v.toString(16);
  }
  return txtHash;
}

function determinarRolUsuario_(correo) {
  const c = correo.toLowerCase();
  if (CONFIG.DOMINIOS_MAESTRO.some(d => c.endsWith(d))) return 'maestro';
  if (CONFIG.DOMINIOS_ALUMNO.some(d => c.endsWith(d)))  return 'alumno';
  return 'denegado';
}

function registrarUsuarioCustom(correo, codigo, contrasenia) {
  try {
    const ss    = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = obtenerOCrearHoja_(ss, CONFIG.SHEET_USUARIOS);
    const data  = sheet.getDataRange().getValues();
    const cL    = String(correo).trim().toLowerCase();
    const coL   = String(codigo).trim().toLowerCase();

    const rol = determinarRolUsuario_(cL);
    if (rol === 'denegado') {
      return {
        exito: false,
        mensaje: 'El correo no pertenece a un dominio autorizado (@alumnos, @academicos, @cucei, @administrativos).'
      };
    }

    for (let i = 1; i < data.length; i++) {
      if (String(data[i][1]).toLowerCase() === cL)  return { exito: false, mensaje: 'El correo ya está registrado.' };
      if (String(data[i][2]).toLowerCase() === coL) return { exito: false, mensaje: 'El código ya está registrado.' };
    }

    const passHash = hashPassword_(contrasenia);
    const nuevoId  = `USR-${Date.now()}`;
    sheet.appendRow([nuevoId, cL, coL, passHash, rol, '', new Date()]);
    registrarLog_(cL, 'REGISTRO_USUARIO', `Rol asignado: ${rol}`);

    return { exito: true, mensaje: 'Registro exitoso.' };
  } catch (e) {
    return { exito: false, mensaje: 'Error interno: ' + e.toString() };
  }
}

function loginUsuarioCustom(identificador, contrasenia) {
  try {
    const ss       = SpreadsheetApp.getActiveSpreadsheet();
    const sheet    = obtenerOCrearHoja_(ss, CONFIG.SHEET_USUARIOS);
    const data     = sheet.getDataRange().getValues();
    const idLower  = String(identificador).trim().toLowerCase();
    const passHash = hashPassword_(contrasenia);

    for (let i = 1; i < data.length; i++) {
      const correo = String(data[i][1]).toLowerCase();
      const codigo = String(data[i][2]).toLowerCase();
      const hash   = String(data[i][3]);

      if (correo === idLower || codigo === idLower) {
        if (hash === passHash) {
          return {
            exito: true,
            usuario: {
              accesoConcedido: true,
              correo:          data[i][1],
              codigo:          data[i][2],
              rol:             data[i][4],
              nombre:          data[i][5] || ''
            }
          };
        }
        return { exito: false, mensaje: 'Contraseña incorrecta.' };
      }
    }
    return { exito: false, mensaje: 'Usuario no encontrado.' };
  } catch (e) {
    return { exito: false, mensaje: 'Error interno: ' + e.toString() };
  }
}

function registrarReserva(datos, sesionCorreo) {
  try {
    // Validación básica
    const validacion = validarDatosReserva_(datos);
    if (!validacion.valido) return { exito: false, mensaje: validacion.mensaje };

    const ss             = SpreadsheetApp.getActiveSpreadsheet();
    const sheetReservas  = obtenerOCrearHoja_(ss, CONFIG.SHEET_RESERVAS);
    const nombreCompleto = `${datos.nombres} ${datos.apellidoPaterno} ${datos.apellidoMaterno}`.trim();
    const odsTexto       = (datos.ods && datos.ods.length > 0)
      ? datos.ods.join(', ')
      : 'Ninguno seleccionado';

    const reservasCreadas = [];
    const fechasConHorario = datos.fechasConHorario || datos.fechas.map(f => ({
      fecha:       f,
      horaEntrada: datos.horaEntrada,
      horaSalida:  datos.horaSalida
    }));

    // Si es reenvío de devuelta, marcar la anterior como reemplazada
    if (datos.reservaDevueltaId) {
      marcarReservaComoReemplazada_(sheetReservas, datos.reservaDevueltaId);
    }

    // Subir archivos a Drive y obtener URLs
    let archivosUrls = [];
    if (datos.archivos && datos.archivos.length > 0) {
      archivosUrls = subirArchivosADrive_(datos.archivos, datos.correo);
    }
    const archivosStr = archivosUrls.length > 0
      ? JSON.stringify(archivosUrls)
      : '';

    // Crear una reserva por cada fecha/horario
    for (const fh of fechasConHorario) {
      const colision = verificarColision_(
        sheetReservas, datos.coworking, fh.fecha, fh.horaEntrada, fh.horaSalida
      );
      if (colision.hayColision) {
        return {
          exito: false,
          mensaje: `Conflicto de horario para el ${formatearFechaLegible_(fh.fecha)}: ${colision.mensaje}`
        };
      }

      const idReserva = generarIdReserva_();
      sheetReservas.appendRow([
        idReserva,                        // 0  ID_Reserva
        datos.correo,                     // 1  Correo
        fh.fecha,                         // 2  Fecha
        fh.horaEntrada,                   // 3  Hora_Inicio
        fh.horaSalida,                    // 4  Hora_Fin
        CONFIG.ESTADOS.PENDIENTE,         // 5  Estado
        datos.coworking,                  // 6  Espacio
        nombreCompleto,                   // 7  Nombre_Completo
        datos.codigo,                     // 8  Codigo
        datos.telefono,                   // 9  Telefono
        datos.extension || 'N/A',         // 10 Extension
        datos.rol,                        // 11 Rol
        datos.actividad,                  // 12 Actividad
        datos.descripcionActividad,       // 13 Descripcion
        odsTexto,                         // 14 ODS
        new Date(),                       // 15 Fecha_Registro
        '',                               // 16 Motivo_Rechazo
        archivosStr,                      // 17 Archivos_URLs
        datos.tipoActividad || '',        // 18 Tipo_Actividad
        datos.correoResponsable || ''     // 19 Correo_Responsable
      ]);
      reservasCreadas.push(idReserva);
    }

    registrarLog_(sesionCorreo, 'RESERVA_CREADA',
      `IDs: ${reservasCreadas.join(', ')} | Coworking: ${datos.coworking}`);

    // Enviar correos
    enviarCorreoConfirmacion_(datos, nombreCompleto, odsTexto, reservasCreadas, fechasConHorario);
    notificarNuevaReservaAdmin_(datos, nombreCompleto, reservasCreadas, fechasConHorario);

    try { procesarRegistroLegacy_(datos, nombreCompleto, odsTexto, fechasConHorario); } catch (e) { /* ignore */ }

    return {
      exito:   true,
      mensaje: `¡Reserva creada con éxito! ID(s): ${reservasCreadas.join(', ')}. Estado: En espera de aprobación.`
    };
  } catch (error) {
    registrarLog_('SISTEMA', 'ERROR_RESERVA', error.toString());
    return { exito: false, mensaje: 'Error al procesar reserva: ' + error.toString() };
  }
}

function marcarReservaComoReemplazada_(sheet, idReservaDevuelta) {

function verificarColision_(sheet, coworking, fecha, horaInicio, horaFin) {
  const ultimaFila = sheet.getLastRow();
  if (ultimaFila <= 1) return { hayColision: false };

  const registros  = sheet.getRange(2, 1, ultimaFila - 1, 7).getValues();
  const tz         = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
  const inicioNuevo = convertirHoraAMinutos_(horaInicio);
  const finNuevo    = convertirHoraAMinutos_(horaFin);

  for (const fila of registros) {
    const [, , fechaEx, hiEx, hfEx, estadoEx, cwEx] = fila;

    if (estadoEx !== CONFIG.ESTADOS.PENDIENTE && estadoEx !== CONFIG.ESTADOS.ACEPTADA) continue;
    if (String(cwEx).trim().toLowerCase() !== coworking.trim().toLowerCase()) continue;

    let fechaNorm;
    try {
      fechaNorm = fechaEx instanceof Date
        ? Utilities.formatDate(fechaEx, tz, 'yyyy-MM-dd')
        : String(fechaEx).trim();
    } catch (e) { continue; }

    if (fechaNorm !== fecha) continue;

    const inicioEx = convertirHoraAMinutos_(String(hiEx).substring(0, 5));
    const finEx    = convertirHoraAMinutos_(String(hfEx).substring(0, 5));

    if (inicioNuevo < finEx && finNuevo > inicioEx) {
      return { hayColision: true, mensaje: 'Superposición de horario con otra reserva existente.' };
    }
  }
  return { hayColision: false };
}

function obtenerDashboardAdmin(adminCorreo) {
  try {
    if (!esAdmin_(adminCorreo)) {
      return { exito: false, mensaje: 'Permisos insuficientes.' };
    }

    const ss            = SpreadsheetApp.getActiveSpreadsheet();
    const sheetReservas = obtenerOCrearHoja_(ss, CONFIG.SHEET_RESERVAS);
    const ultimaFila    = sheetReservas.getLastRow();

    if (ultimaFila <= 1) {
      return {
        exito: true,
        reservas: [],
        metricas: { total: 0, pendientes: 0, aceptadas: 0, rechazadas: 0 },
        graficaEstados:  { labels: ['Pendiente','Aceptada','Rechazada'], datos: [0,0,0] },
        graficaEspacios: { labels: [], datos: [] }
      };
    }

    // Leer 20 columnas para incluir nuevos campos
    const numCols = 20;
    const datos   = sheetReservas.getRange(2, 1, ultimaFila - 1, numCols).getValues();
    const tz      = ss.getSpreadsheetTimeZone();
    const reservas    = [];
    let pendientes = 0, aceptadas = 0, rechazadas = 0;
    const espaciosConteo = {};

    for (const fila of datos) {
      const idReserva = fila[CONFIG.COL.ID_RESERVA];
      if (!idReserva) continue;

      const estado    = String(fila[CONFIG.COL.ESTADO]);
      const coworking = String(fila[CONFIG.COL.COWORKING]).trim();
      let fechaStr    = fila[CONFIG.COL.FECHA] instanceof Date
        ? Utilities.formatDate(fila[CONFIG.COL.FECHA], tz, 'yyyy-MM-dd')
        : String(fila[CONFIG.COL.FECHA]).trim();

      switch (estado) {
        case CONFIG.ESTADOS.PENDIENTE:  pendientes++;  break;
        case CONFIG.ESTADOS.ACEPTADA:   aceptadas++;   break;
        case CONFIG.ESTADOS.RECHAZADA:  rechazadas++;  break;
      }

      espaciosConteo[coworking] = (espaciosConteo[coworking] || 0) + 1;

      // Parsear archivos URLs
      let archivosUrls = [];
      try {
        const archStr = String(fila[CONFIG.COL.ARCHIVOS_URLS] || '');
        if (archStr && archStr !== '') archivosUrls = JSON.parse(archStr);
      } catch (e) { /* ignore */ }

      reservas.push({
        idReserva:         String(idReserva),
        correo:            String(fila[CONFIG.COL.CORREO]),
        fecha:             fechaStr,
        horaInicio:        formatearHoraSegura_(fila[CONFIG.COL.HORA_INICIO]),
        horaFin:           formatearHoraSegura_(fila[CONFIG.COL.HORA_FIN]),
        estado:            estado,
        coworking:         coworking,
        nombreCompleto:    String(fila[CONFIG.COL.NOMBRE_COMPLETO]),
        codigo:            String(fila[CONFIG.COL.CODIGO]),
        telefono:          String(fila[CONFIG.COL.TELEFONO]),
        extension:         String(fila[CONFIG.COL.EXTENSION]),
        rol:               String(fila[CONFIG.COL.ROL]),
        actividad:         String(fila[CONFIG.COL.ACTIVIDAD]),
        descripcion:       String(fila[CONFIG.COL.DESCRIPCION]),
        ods:               String(fila[CONFIG.COL.ODS]),
        fechaRegistro:     fila[CONFIG.COL.FECHA_REGISTRO] instanceof Date
          ? Utilities.formatDate(fila[CONFIG.COL.FECHA_REGISTRO], tz, 'dd/MM/yyyy HH:mm')
          : String(fila[CONFIG.COL.FECHA_REGISTRO]),
        motivoRechazo:     String(fila[CONFIG.COL.MOTIVO_RECHAZO] || ''),
        archivosUrls:      archivosUrls,
        tipoActividad:     String(fila[CONFIG.COL.TIPO_ACTIVIDAD]   || ''),
        correoResponsable: String(fila[CONFIG.COL.CORREO_RESPONSABLE] || '')
      });
    }

    reservas.sort((a, b) => {
      const p = { 'Pendiente': 0, 'Devuelta': 1, 'Aceptada': 2, 'Rechazada': 3 };
      if ((p[a.estado] ?? 4) !== (p[b.estado] ?? 4)) return (p[a.estado] ?? 4) - (p[b.estado] ?? 4);
      return b.fecha.localeCompare(a.fecha);
    });

    registrarLog_(adminCorreo, 'DASHBOARD_CONSULTADO', `Total: ${reservas.length}`);

    return {
      exito:   true,
      reservas,
      metricas: { total: reservas.length, pendientes, aceptadas, rechazadas },
      graficaEstados:  {
        labels: ['Pendiente', 'Aceptada', 'Rechazada'],
        datos:  [pendientes, aceptadas, rechazadas]
      },
      graficaEspacios: {
        labels: Object.keys(espaciosConteo),
        datos:  Object.values(espaciosConteo)
      }
    };
  } catch (error) {
    return { exito: false, mensaje: error.toString() };
  }
}

function actualizarEstadoReserva(idReserva, nuevoEstado, adminCorreo, motivoRechazo) {
  try {
    if (!esAdmin_(adminCorreo)) return { exito: false, mensaje: 'Permisos insuficientes.' };

    const ss            = SpreadsheetApp.getActiveSpreadsheet();
    const sheetReservas = obtenerOCrearHoja_(ss, CONFIG.SHEET_RESERVAS);
    const ultimaFila    = sheetReservas.getLastRow();
    if (ultimaFila <= 1) return { exito: false, mensaje: 'No hay reservas.' };

    const numCols = 20;
    const datos   = sheetReservas.getRange(2, 1, ultimaFila - 1, numCols).getValues();

    for (let i = 0; i < datos.length; i++) {
      if (String(datos[i][CONFIG.COL.ID_RESERVA]).trim() !== String(idReserva).trim()) continue;

      const estadoActual = datos[i][CONFIG.COL.ESTADO];

      // Reservas aceptadas no se pueden cambiar
      if (estadoActual === CONFIG.ESTADOS.ACEPTADA) {
        return { exito: false, mensaje: '🔒 Esta reserva ya fue Aceptada y no puede modificarse.' };
      }

      const fila = i + 2; // +1 encabezado, +1 base 1

      // Actualizar estado
      sheetReservas.getRange(fila, CONFIG.COL.ESTADO + 1).setValue(nuevoEstado);

      // Guardar motivo de rechazo si aplica
      if (motivoRechazo && (nuevoEstado === CONFIG.ESTADOS.RECHAZADA || nuevoEstado === CONFIG.ESTADOS.DEVUELTA)) {
        sheetReservas.getRange(fila, CONFIG.COL.MOTIVO_RECHAZO + 1).setValue(motivoRechazo);
      }

      registrarLog_(adminCorreo,
        `RESERVA_${nuevoEstado.toUpperCase()}`,
        `ID: ${idReserva} | Admin: ${adminCorreo}`
      );

      // Enviar notificación al usuario
      const correoUsuario = String(datos[i][CONFIG.COL.CORREO]);
      enviarNotificacionCambioEstado_(correoUsuario, idReserva, nuevoEstado, datos[i], motivoRechazo);

      // Crear evento en Calendar si fue aceptada
      if (nuevoEstado === CONFIG.ESTADOS.ACEPTADA) {
        crearEventoCalendario_(datos[i], adminCorreo);
      }

      return {
        exito:   true,
        mensaje: `✅ Reserva actualizada a "${nuevoEstado}" correctamente.`
      };
    }

    return { exito: false, mensaje: 'Reserva no encontrada.' };
  } catch (error) {
    return { exito: false, mensaje: error.toString() };
  }
}

function buscarReservasPorCorreo(correoBusqueda, adminCorreo) {
  try {
    if (!esAdmin_(adminCorreo)) return { exito: false, mensaje: 'Permisos insuficientes.' };

    const ss            = SpreadsheetApp.getActiveSpreadsheet();
    const sheetReservas = obtenerOCrearHoja_(ss, CONFIG.SHEET_RESERVAS);
    const ultimaFila    = sheetReservas.getLastRow();
    if (ultimaFila <= 1) return { exito: true, reservas: [], mensaje: 'No hay reservas.' };

    const datos      = sheetReservas.getRange(2, 1, ultimaFila - 1, 16).getValues();
    const tz         = ss.getSpreadsheetTimeZone();
    const resultados = [];
    const busqueda   = correoBusqueda.trim().toLowerCase();

    for (const fila of datos) {
      if (!String(fila[1]).toLowerCase().includes(busqueda)) continue;
      let fStr = fila[2] instanceof Date
        ? Utilities.formatDate(fila[2], tz, 'yyyy-MM-dd')
        : String(fila[2]).trim();
      resultados.push({
        idReserva:  String(fila[0]),
        correo:     String(fila[1]),
        fecha:      fStr,
        horaInicio: formatearHoraSegura_(fila[3]),
        horaFin:    formatearHoraSegura_(fila[4]),
        estado:     String(fila[5]),
        coworking:  String(fila[6])
      });
    }

    registrarLog_(adminCorreo, 'BUSQUEDA_REALIZADA', `"${correoBusqueda}"`);
    return {
      exito:    true,
      reservas: resultados,
      mensaje:  resultados.length > 0
        ? `Se encontraron ${resultados.length} reserva(s).`
        : 'No se encontraron reservas para ese correo.'
    };
  } catch (error) {
    return { exito: false, mensaje: error.toString() };
  }
}

// GOOGLE DRIVE – Subida de archivos
// ─────────────────────────────────────────────────────────────

function subirArchivosADrive_(archivos, correoUsuario) {
  const urls = [];
  try {
    const folder = obtenerOCrearCarpetaDrive_();
    const subFolder = obtenerOCrearSubcarpeta_(folder, correoUsuario);

    for (const archivo of archivos) {
      try {
        if (!archivo.base64 || !archivo.nombre) continue;

        const blob = Utilities.newBlob(
          Utilities.base64Decode(archivo.base64),
          archivo.tipo || 'application/octet-stream',
          archivo.nombre
        );

        const driveFile = subFolder.createFile(blob);
        driveFile.setSharing(
          DriveApp.Access.ANYONE_WITH_LINK,
          DriveApp.Permission.VIEW
        );

        urls.push({
          nombre: archivo.nombre,
          url:    driveFile.getUrl(),
          id:     driveFile.getId()
        });
      } catch (eArch) {
        registrarLog_('SISTEMA', 'ERROR_SUBIDA_ARCHIVO',
          `Archivo: ${archivo.nombre} | Error: ${eArch.toString()}`);
      }
    }
  } catch (e) {
    registrarLog_('SISTEMA', 'ERROR_DRIVE', e.toString());
  }
  return urls;
}

function obtenerOCrearCarpetaDrive_() {
  const carpetas = DriveApp.getFoldersByName(CONFIG.DRIVE_FOLDER_NAME);
  return carpetas.hasNext()
    ? carpetas.next()
    : DriveApp.createFolder(CONFIG.DRIVE_FOLDER_NAME);
}

function obtenerOCrearSubcarpeta_(padre, nombre) {
  const nombreLimpio = String(nombre).replace(/[^a-zA-Z0-9@._-]/g, '_');
  const subs = padre.getFoldersByName(nombreLimpio);
  return subs.hasNext()
    ? subs.next()
    : padre.createFolder(nombreLimpio);
}

function crearEventoCalendario_(filaReserva, adminCorreo) {
  try {
    const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();

    let fStr = filaReserva[CONFIG.COL.FECHA] instanceof Date
      ? Utilities.formatDate(filaReserva[CONFIG.COL.FECHA], tz, 'yyyy-MM-dd')
      : String(filaReserva[CONFIG.COL.FECHA]).trim();

    // Normalizar formato de fecha
    if (fStr.includes('/')) {
      const partes = fStr.split('/');
      fStr = `${partes[2]}-${partes[1].padStart(2,'0')}-${partes[0].padStart(2,'0')}`;
    }

    const horaInicio = formatearHoraSegura_(filaReserva[CONFIG.COL.HORA_INICIO]);
    const horaFin    = formatearHoraSegura_(filaReserva[CONFIG.COL.HORA_FIN]);
    const inicio     = new Date(`${fStr}T${horaInicio}:00`);
    const fin        = new Date(`${fStr}T${horaFin}:00`);

    if (isNaN(inicio.getTime()) || isNaN(fin.getTime())) {
      registrarLog_('SISTEMA', 'ERROR_CALENDAR', `Fechas inválidas: ${fStr} ${horaInicio}-${horaFin}`);
      return;
    }

    const correoSolicitante = String(filaReserva[CONFIG.COL.CORREO]);
    const nombre            = String(filaReserva[CONFIG.COL.NOMBRE_COMPLETO]);
    const coworking         = String(filaReserva[CONFIG.COL.COWORKING]);
    const actividad         = String(filaReserva[CONFIG.COL.ACTIVIDAD]);
    const descripcion       = String(filaReserva[CONFIG.COL.DESCRIPCION]);
    const correoResponsable = String(filaReserva[CONFIG.COL.CORREO_RESPONSABLE] || '');

    // Construir lista de invitados
    const invitados = [correoSolicitante];
    if (adminCorreo && adminCorreo !== correoSolicitante) invitados.push(adminCorreo);
    if (CONFIG.CORREO_COORDINADOR_1) invitados.push(CONFIG.CORREO_COORDINADOR_1);
    if (CONFIG.CORREO_COORDINADOR_2) invitados.push(CONFIG.CORREO_COORDINADOR_2);
    if (correoResponsable && !invitados.includes(correoResponsable)) {
      invitados.push(correoResponsable);
    }

    const titulo      = `✅ Reserva Confirmada: ${coworking} – ${actividad}`;
    const descripEvento = `Reserva confirmada en ${coworking}.\n\nSolicitante: ${nombre}\nCorreo: ${correoSolicitante}\nActividad: ${actividad}\n\n${descripcion}`;

    CalendarApp.getDefaultCalendar().createEvent(titulo, inicio, fin, {
      description:  descripEvento,
      guests:       invitados.join(','),
      sendInvites:  true
    });

    registrarLog_('SISTEMA', 'CALENDAR_CREADO',
      `Evento en ${coworking} para ${correoSolicitante} | Invitados: ${invitados.join(', ')}`);

  } catch (e) {
    registrarLog_('SISTEMA', 'ERROR_CALENDAR', e.toString());
  }
}

/** Notificación al usuario sobre el cambio de estado de su reserva */
function enviarNotificacionCambioEstado_(correoUsuario, idReserva, nuevoEstado, filaReserva, motivoRechazo) {
  try {
    const esAprobada = nuevoEstado === CONFIG.ESTADOS.ACEPTADA;
    const esDevuelta = nuevoEstado === CONFIG.ESTADOS.DEVUELTA;
    const coworking  = String(filaReserva[CONFIG.COL.COWORKING]);
    const nombre     = String(filaReserva[CONFIG.COL.NOMBRE_COMPLETO]).split(' ')[0];
    const actividad  = String(filaReserva[CONFIG.COL.ACTIVIDAD]);

    let estadoColor  = '#ef4444';
    let estadoIcono  = '❌';
    let titulo       = 'Solicitud Rechazada';
    let subtitulo    = 'Tu solicitud de reserva no fue aprobada';
    let mensajePpal  = `Lamentamos informarte que tu solicitud de reserva en ${coworking} ha sido rechazada.`;
    let mensajeAdc   = null;

    if (esAprobada) {
      estadoColor  = '#10b981';
      estadoIcono  = '✅';
      titulo       = '¡Reserva Aprobada!';
      subtitulo    = 'Tu solicitud de reserva ha sido confirmada';
      mensajePpal  = `¡Excelente noticia! Tu solicitud de reserva en ${coworking} ha sido aprobada. Revisa tu calendario para ver el evento creado.`;
    } else if (esDevuelta) {
      estadoColor  = '#8b5cf6';
      estadoIcono  = '↩️';
      titulo       = 'Solicitud Devuelta para Edición';
      subtitulo    = 'El administrador ha devuelto tu solicitud para que la corrijas';
      mensajePpal  = `Tu solicitud de reserva en ${coworking} ha sido devuelta. Por favor, revisa el motivo e ingresa al sistema para corregirla y reenviarla.`;
    }

    if (motivoRechazo) {
      mensajeAdc = `Motivo: ${motivoRechazo}`;
    }

    let fechaStr = '';
    try {
      const tz  = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
      const fRaw = filaReserva[CONFIG.COL.FECHA];
      const fS   = fRaw instanceof Date
        ? Utilities.formatDate(fRaw, tz, 'yyyy-MM-dd')
        : String(fRaw).trim();
      const [y, m, d] = fS.split('-');
      fechaStr = `${d}/${m}/${y}`;
    } catch (e) { /* ignore */ }

    const html = generarPlantillaCorporativa_({
      titulo,
      subtitulo,
      nombreUsuario:    nombre,
      estado:           nuevoEstado,
      estadoColor,
      estadoIcono,
      mensajePrincipal: mensajePpal,
      mensajeAdicional: mensajeAdc,
      detalles: [
        { etiqueta: 'ID Reserva',  valor: idReserva },
        { etiqueta: 'Espacio',     valor: coworking },
        { etiqueta: 'Actividad',   valor: actividad },
        { etiqueta: 'Fecha',       valor: fechaStr },
        { etiqueta: 'Horario',     valor: `${formatearHoraSegura_(filaReserva[CONFIG.COL.HORA_INICIO])} – ${formatearHoraSegura_(filaReserva[CONFIG.COL.HORA_FIN])}` }
      ]
    });

    const asunto = esAprobada
      ? `✅ Reserva Aprobada – ${coworking}`
      : esDevuelta
        ? `↩️ Solicitud Devuelta para Edición – ${coworking}`
        : `❌ Solicitud Rechazada – ${coworking}`;

    MailApp.sendEmail({ to: correoUsuario, subject: asunto, htmlBody: html });
  } catch (e) {
    registrarLog_('SISTEMA', 'ERROR_CORREO_ESTADO', e.toString());
  }
}

function convertirHoraAMinutos_(horaStr) {

function validarDatosReserva_(datos) {
  const requeridos = ['correo','nombres','apellidoPaterno','coworking'];
  for (const c of requeridos) {
    if (!datos[c] || String(datos[c]).trim() === '') {
      return { valido: false, mensaje: `El campo "${c}" es obligatorio.` };
    }
  }
  const fch = datos.fechasConHorario || [];
  if (fch.length === 0 && (!datos.fechas || datos.fechas.length === 0)) {

function formatearHoraSegura_(valor) {
  if (!valor) return '00:00';
  if (valor instanceof Date) {
    return Utilities.formatDate(
      valor,
      SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(),
      'HH:mm'
    );
  }
  const str   = String(valor).trim();
  const match = str.match(/(\d+):(\d+)\s*(AM|PM)?/i);
  if (match) {
    let h = parseInt(match[1], 10);
    if (match[3]?.toUpperCase() === 'PM' && h < 12) h += 12;
    if (match[3]?.toUpperCase() === 'AM' && h === 12) h = 0;
    return `${String(h).padStart(2,'0')}:${match[2]}`;
  }
  return str.substring(0, 5);
}
