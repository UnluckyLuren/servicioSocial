/**
 * Genera un ID único para la base de datos (Ej: USR-8f7a2b)
 */
function generarIdUnico(prefijo = 'USR-') {
  return prefijo + Math.random().toString(36).substr(2, 6);
}

/**
 * Función para autenticar al usuario contra la hoja "Usuarios".
 * @param {string} codigo - Código del estudiante/docente.
 * @param {string} nip - Contraseña o NIP.
 */
function autenticarUsuario(codigo, nip) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Usuarios");
    const data = sheet.getDataRange().getValues();
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      // Sanitizamos los datos de entrada para evitar fallos de coincidencia
      if (row[1].toString().trim() === codigo.toString().trim() && row[5].toString().trim() === nip.toString().trim()) {
        return {
          exito: true,
          usuario: {
            idInterno: row[0].toString(), 
            code: row[1].toString().trim(),
            name: row[2].toString().trim(),
            // ELIMINAMOS ESPACIOS OCULTOS DEL ROL AL INICIAR SESIÓN
            role: row[3].toString().trim(),
            email: row[4].toString().trim(),
            photo: row[6] || null,
            career: "Ingeniería en Computación (INCO)",
            gpa: 90.5,
            credits: 200,
            totalCredits: 380,
            integralHours: 40,
            requiredHours: 60,
            status: "Activo Regular",
            cycle: "2026B"
          }
        };
      }
    }
    return { exito: false, mensaje: "Código o NIP incorrectos. Verifica tus datos." };
  } catch (error) {
    return { exito: false, mensaje: "Error de servidor: " + error.toString() };
  }
}

/**
 * Función para registrar un nuevo usuario desde el Frontend.
 */
function registrarUsuario(datosUsuario) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Usuarios");
    const nuevoId = generarIdUnico('USR-'); 
    
    // Verificación simple: ¿Ya existe el código?
    const data = sheet.getDataRange().getValues();
    for(let i = 1; i < data.length; i++){
       if(data[i][1].toString() === datosUsuario.code.toString()){
          return { exito: false, mensaje: "El código ingresado ya está registrado." };
       }
    }

    // Insertamos la fila (debe coincidir con las columnas de Google Sheets)
    sheet.appendRow([
      nuevoId,               // A: ID_Usuario
      datosUsuario.code,     // B: Codigo
      datosUsuario.name,     // C: Nombre
      datosUsuario.role,     // D: Rol
      datosUsuario.email,    // E: Email
      datosUsuario.nip,      // F: NIP
      datosUsuario.photo     // G: Foto
    ]);
    
    return { 
      exito: true, 
      usuario: { ...datosUsuario, idInterno: nuevoId, photo: datosUsuario.photo || null } 
    };
    
  } catch(error) {
    return { exito: false, mensaje: "Error al registrar: " + error.toString() };
  }
}

/**
 * Carga Tickets, Avisos y Justificantes al iniciar sesión con límite de seguridad.
 * @param {string} codigoUsuario - Código UdG del usuario logueado.
 * @param {string} rol - Rol del usuario ('Estudiante', 'Docente' o 'Coordinacion').
 * @param {number} limiteMaximo - Cantidad máxima de registros a retornar (por defecto 100).
 */
function obtenerDatosPlataforma(codigoUsuario, rol, limiteMaximo = 100) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let tickets = [];
    let avisos = [];
    let justificantes = [];

    const rolLimpio = String(rol).trim();
    const codigoLimpio = String(codigoUsuario).trim();

    // 1. Cargar Tickets
    const hojaTickets = ss.getSheetByName("Tickets");
    if (hojaTickets) {
      const dataTickets = hojaTickets.getDataRange().getValues();
      for (let i = 1; i < dataTickets.length; i++) {
        const fila = dataTickets[i];
        if (fila[0] && String(fila[0]).trim() !== "") {
          if (rolLimpio === 'Coordinacion' || String(fila[1]).trim() === codigoLimpio) {
            let fechaTicket = fila[7] instanceof Date ? fila[7].toLocaleDateString('es-MX') : String(fila[7]);
            tickets.push({
              id: String(fila[0]), 
              code: String(fila[1]).trim(), 
              student: String(fila[2]),
              subject: String(fila[3]), 
              details: String(fila[4]), 
              status: String(fila[5]),
              reply: String(fila[6]), 
              date: fechaTicket
            });
          }
        }
      }
    }

    // 2. Cargar Avisos
    const hojaAvisos = ss.getSheetByName("Avisos");
    if (hojaAvisos) {
      const dataAvisos = hojaAvisos.getDataRange().getValues();
      for (let i = 1; i < dataAvisos.length; i++) {
        const fila = dataAvisos[i];
        if (fila[0] && String(fila[0]).trim() !== "") {
          let fechaAviso = fila[2] instanceof Date ? fila[2].toLocaleDateString('es-MX') : String(fila[2]);
          avisos.push({
            id: String(fila[0]), 
            title: String(fila[1]), 
            date: fechaAviso, 
            cat: String(fila[3]), 
            urgent: String(fila[4]).trim() === 'Sí', 
            desc: String(fila[5])
          });
        }
      }
    }

    // 3. Cargar Justificantes
    const hojaJustificantes = ss.getSheetByName("Justificantes");
    if (hojaJustificantes) {
      const dataJus = hojaJustificantes.getDataRange().getValues();
      for (let i = 1; i < dataJus.length; i++) {
        const fila = dataJus[i];
        if (fila[0] && String(fila[0]).trim() !== "") {
          if (rolLimpio === 'Coordinacion' || String(fila[1]).trim() === codigoLimpio) {
            let fechaJus = fila[9] instanceof Date ? fila[9].toLocaleDateString('es-MX') : String(fila[9]);
            justificantes.push({
              id: String(fila[0]),
              code: String(fila[1]).trim(),
              student: String(fila[2]),
              reason: String(fila[3]),
              teachers: String(fila[4]),
              startDate: String(fila[5]),
              endDate: String(fila[6]),
              fileName: String(fila[7]),
              status: String(fila[8]),
              date: fechaJus,
              email: String(fila[10] || ''),     // NUEVO
              phone: String(fila[11] || ''),     // NUEVO
              adminPdfUrl: String(fila[12] || '')// NUEVO
            });
          }
        }
      }
    }

    // 4. Cargar Horas Integrales
    let horas = [];
    const hojaHoras = ss.getSheetByName("HorasIntegrales");
    if (hojaHoras) {
      const dataHoras = hojaHoras.getDataRange().getValues();
      for (let i = 1; i < dataHoras.length; i++) {
        const fila = dataHoras[i];
        if (fila[0] && String(fila[0]).trim() !== "") {
          if (rolLimpio === 'Coordinacion' || String(fila[1]).trim() === codigoLimpio) {
           horas.push({
              id: String(fila[0]),
              code: String(fila[1]).trim(),
              student: String(fila[2]),
              confirmo64: String(fila[3]) === 'Sí',
              files: String(fila[4]),
              status: String(fila[5]),
              date: String(fila[6]),
              horasAprobadas: Number(fila[7]) || 0 // NUEVA LÍNEA: Lee la columna H
            });
          }
        }
      }
    }

    // 5. Cargar Constancias 
    let constancias = [];
    const hojaConstancias = ss.getSheetByName("Constancias");
    if (hojaConstancias) {
      const dataConstancias = hojaConstancias.getDataRange().getValues();
      for (let i = 1; i < dataConstancias.length; i++) {
        const fila = dataConstancias[i];
        if (fila[0] && String(fila[0]).trim() !== "") {
          if (rolLimpio === 'Coordinacion' || String(fila[1]).trim() === codigoLimpio) {
            let fechaSol = fila[6] instanceof Date ? fila[6].toLocaleDateString('es-MX') : String(fila[6]);
            constancias.push({
              id: String(fila[0]),
              code: String(fila[1]).trim(),
              student: String(fila[2]),
              email: String(fila[3]),
              requiresDetails: String(fila[4]),
              status: String(fila[5]),
              date: fechaSol,
              adminData: fila[7] ? String(fila[7]) : "{}"
            });
          }
        }
      }
    }
    
   return { 
      exito: true, 
      tickets: tickets.reverse().slice(0, limiteMaximo), 
      avisos: avisos.reverse().slice(0, limiteMaximo),
      justificantes: justificantes.reverse().slice(0, limiteMaximo),
      horas: horas.reverse() 
    };
  } catch (error) {
    return { exito: false, mensaje: "Error cargando la base de datos: " + error.toString() };
  }
}

/**
 * Guarda la respuesta del administrador, cambia el estatus a "Atendido" y notifica al alumno por correo.
 */
function responderTicketAdmin(ticketId, respuestaAdmin) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheetTickets = ss.getSheetByName("Tickets");
    const dataTickets = sheetTickets.getDataRange().getValues();
    
    let studentCode = "";
    let studentName = "";
    let subject = "";
    let details = "";
    let filaActualizar = -1;

    // 1. Buscar el ticket y actualizarlo
    for (let i = 1; i < dataTickets.length; i++) {
      if (dataTickets[i][0].toString() === ticketId.toString()) {
        filaActualizar = i + 1;
        studentCode = dataTickets[i][1].toString();
        studentName = dataTickets[i][2].toString();
        subject = dataTickets[i][3].toString();
        details = dataTickets[i][4].toString();
        
        // Columna 6 = Estatus, Columna 7 = Respuesta
        sheetTickets.getRange(filaActualizar, 6).setValue("Atendido");
        sheetTickets.getRange(filaActualizar, 7).setValue(respuestaAdmin);
        break;
      }
    }

    if (filaActualizar === -1) {
      return { exito: false, mensaje: "El ticket no se encontró en la base de datos." };
    }

    // 2. Buscar el correo del alumno en la hoja Usuarios
    const sheetUsuarios = ss.getSheetByName("Usuarios");
    let studentEmail = "";
    if (sheetUsuarios) {
      const dataUsuarios = sheetUsuarios.getDataRange().getValues();
      for (let j = 1; j < dataUsuarios.length; j++) {
        if (dataUsuarios[j][1].toString() === studentCode) {
          studentEmail = dataUsuarios[j][4].toString(); // Columna E es el Email
          break;
        }
      }
    }

    // 3. Enviar correo al alumno con la respuesta si encontramos su email
    if (studentEmail) {
      const asuntoCorreo = `Respuesta a tu Ticket #${ticketId} - Coordinación INCO`;
      const fechaActual = new Date().toLocaleDateString('es-MX');

      const cuerpoHtml = `
        <div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto; border: 1px solid #e2e8f0; border-radius: 10px; overflow: hidden; box-shadow: 0 4px 6px rgba(0,0,0,0.05);">

            <p style="color: #cbd5e1; font-size: 13px; margin-top: 5px;">Universidad de Guadalajara • CUCEI</p>
          </div>
          
          <div style="padding: 30px; background-color: #ffffff; color: #334155;">
            <h3 style="color: #059669; border-bottom: 2px solid #059669; padding-bottom: 8px; margin-top: 0;">Tu ticket ha sido atendido</h3>
            <p style="font-size: 15px;">Hola <strong>${studentName}</strong>,</p>

/**
 * Registra un nuevo justificante, guarda los datos extra y envía correo al alumno.
 */
function registrarJustificante(datos) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Justificantes");
    const nuevoId = generarIdUnico('JUS-');
    const fechaActual = new Date().toLocaleDateString('es-MX');

    let fileUrl = "Sin archivo";
    if (datos.fileData && datos.fileName) {
      const splitBase = datos.fileData.split(',');
      const type = splitBase[0].split(';')[0].replace('data:', '');
      const byteCharacters = Utilities.base64Decode(splitBase[1]);
      const extension = datos.fileName.split('.').pop();
      const nombreLimpio = datos.student.replace(/\s+/g, '');
      const fechaLimpia = fechaActual.replace(/\//g, '-');
      const nombreInteligente = `${datos.code}_${nombreLimpio}_${datos.reason}_${fechaLimpia}.${extension}`;
      
      const blob = Utilities.newBlob(byteCharacters, type, nombreInteligente);
      const folderIdJustificantes = "1FYLyqOiFs90s337WMALx763R5iTxg9LW"; 
      const folder = DriveApp.getFolderById(folderIdJustificantes);
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fileUrl = file.getUrl();
    }

    // Insertar fila (Asegúrate de tener 13 columnas en Google Sheets)
    sheet.appendRow([
      nuevoId, datos.code, datos.student, datos.reason, datos.teachers, 
      datos.startDate, datos.endDate || datos.startDate, fileUrl, 
      "Pendiente", fechaActual, datos.email, datos.phone, ""
    ]);

    // Enviar correo de confirmación al alumno
    if (datos.email) {
      MailApp.sendEmail({
        to: datos.email,
        subject: `Solicitud de Justificante Recibida - #${nuevoId}`,
        htmlBody: `
          <div style="font-family: Arial, sans-serif; padding: 20px; color: #333; max-width: 600px; border: 1px solid #ddd; border-radius: 8px;">
            <h2 style="color: #0f2744;">Solicitud Recibida Exitosamente</h2>
            <p>Hola <strong>${datos.student}</strong>,</p>
            <p>Tu solicitud de justificante por <strong>${datos.reason}</strong> ha sido recibida y está en revisión por la Coordinación INCO.</p>
            <div style="background-color: #f8fafc; padding: 15px; border-left: 4px solid #d97706; margin-bottom: 20px;">
              <p style="margin:0;"><strong>Folio:</strong> #${nuevoId}</p>
              <p style="margin:5px 0 0 0;"><strong>Materias afectadas:</strong> ${datos.teachers}</p>
            </div>
            <p>Recibirás un correo cuando el documento oficial esté listo para recoger.</p>
          </div>
        `
      });
    }

    return { 
      exito: true, 
      justificante: {
        id: nuevoId, code: datos.code, student: datos.student, reason: datos.reason, 
        teachers: datos.teachers, startDate: datos.startDate, endDate: datos.endDate || datos.startDate,
        fileName: fileUrl, status: "Pendiente", date: fechaActual, email: datos.email, phone: datos.phone, adminPdfUrl: ""
      }
    };
  } catch(error) {
    return { exito: false, mensaje: "Error al guardar: " + error.toString() };
  }
}

/**
 * Avanza el estado del Justificante. (Sustituye a dictaminarJustificante)
 * - Generar: Crea el PDF y pasa a 'En Proceso'.
 * - Finalizar: Pasa a 'Finalizada' y manda correo al alumno.
 */
function avanzarJustificante(id, accion) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Justificantes");
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    let rowData = null;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(id).trim()) {
        rowIndex = i + 1;
        rowData = data[i];

/**
 * Envía una nueva solicitud de Horas Integrales con múltiples archivos.
 */
function registrarSolicitudHoras(datos) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("HorasIntegrales");
    if(!sheet) return { exito: false, mensaje: "No existe la hoja HorasIntegrales" };

    const nuevoId = generarIdUnico('HRS-');
    const fechaActual = new Date().toLocaleDateString('es-MX');
    
    // ID DE LA CARPETA DONDE SE GUARDARÁN LAS EVIDENCIAS DE HORAS INTEGRALES
    const folderIdHoras = "15Qc4mLLaDB0TYxonEiqKEmtoAZKF-NNf"; 
    const folder = DriveApp.getFolderById(folderIdHoras);
    
    let urls = [];
    
    // Guardar múltiples archivos
    if (datos.archivos && datos.archivos.length > 0) {
      for (let i = 0; i < datos.archivos.length; i++) {
        let fileObj = datos.archivos[i];
        let splitBase = fileObj.data.split(',');
        let type = splitBase[0].split(';')[0].replace('data:', '');
        let byteCharacters = Utilities.base64Decode(splitBase[1]);
        
        let extension = fileObj.name.split('.').pop();
        let nombreLimpio = datos.student.replace(/\s+/g, '');
        let nombreArchivo = `${datos.code}_${nombreLimpio}_Evidencia${i+1}_${fechaActual.replace(/\//g, '-')}.${extension}`;
        
        let blob = Utilities.newBlob(byteCharacters, type, nombreArchivo);
        let file = folder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        urls.push(file.getUrl());
      }
    }

    const urlsString = urls.join(', ');

    sheet.appendRow([
      nuevoId, datos.code, datos.student, datos.confirma64 ? "Sí" : "No", 
      urlsString, "Pendiente", fechaActual
    ]);

    return { exito: true, solicitud: { id: nuevoId, code: datos.code, student: datos.student, confirmo64: datos.confirma64, files: urlsString, status: "Pendiente", date: fechaActual }};
  } catch(e) {
    return { exito: false, mensaje: "Error al guardar: " + e.toString() };
  }
}

/**
 * Agrega más archivos a una solicitud devuelta, actualiza la confirmación de 64h y los archivos previos que se mantienen.
 */
function agregarArchivosHoras(id, nuevosArchivos, confirma64, urlsMantenidas) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("HorasIntegrales");
    const data = sheet.getDataRange().getValues();
    
    // ID DE LA CARPETA DONDE SE GUARDAN LAS EVIDENCIAS
    const folderIdHoras = "15Qc4mLLaDB0TYxonEiqKEmtoAZKF-NNf"; 
    const folder = DriveApp.getFolderById(folderIdHoras);
    const fechaActual = new Date().toLocaleDateString('es-MX');

    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(id).trim()) {
        
        let nuevasUrls = [];
        
        // Guardamos los archivos nuevos si es que el alumno subió alguno
        if (nuevosArchivos && nuevosArchivos.length > 0) {
          for (let j = 0; j < nuevosArchivos.length; j++) {
            let fileObj = nuevosArchivos[j];
            let splitBase = fileObj.data.split(',');
            let type = splitBase[0].split(';')[0].replace('data:', '');
            let byteCharacters = Utilities.base64Decode(splitBase[1]);
            let extension = fileObj.name.split('.').pop();
            
            let nombreArchivo = `${data[i][1]}_AnexoExtra_${fechaActual.replace(/\//g, '-')}_${Math.floor(Math.random()*1000)}.${extension}`;
            let blob = Utilities.newBlob(byteCharacters, type, nombreArchivo);
            let file = folder.createFile(blob);
            file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
            nuevasUrls.push(file.getUrl());
          }
        }
        
        // Combinar las URLs que el usuario decidió CONSERVAR con las nuevas que acaba de subir
        let arrayFinal = [];
        if (urlsMantenidas && String(urlsMantenidas).trim() !== "") {
          arrayFinal.push(String(urlsMantenidas).trim());
        }
        if (nuevasUrls.length > 0) {
          arrayFinal.push(nuevasUrls.join(', '));
        }
        const stringFinalUrls = arrayFinal.join(', ');
        
        sheet.getRange(i + 1, 4).setValue(confirma64 ? "Sí" : "No");
        sheet.getRange(i + 1, 5).setValue(stringFinalUrls);
        sheet.getRange(i + 1, 6).setValue("Pendiente"); 
        
        return { exito: true, nuevasUrls: stringFinalUrls, confirmo64: confirma64 };
      }
    }
    return { exito: false, mensaje: "Solicitud no encontrada" };
  } catch(e) {
    return { exito: false, mensaje: "Error al actualizar: " + e.toString() };
  }
}

/**
 * El admin aprueba o devuelve la solicitud
 */
function dictaminarHoras(id, nuevoEstatus, horasOtorgadas = 0) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("HorasIntegrales");
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(id).trim()) {
        sheet.getRange(i + 1, 6).setValue(nuevoEstatus); // Actualiza estatus en F
        
        // Si se aprueba, guardamos la cantidad de horas en la columna H
        if (nuevoEstatus === 'Aprobado') {
          sheet.getRange(i + 1, 8).setValue(horasOtorgadas);
        }
        return { exito: true };
      }
    }
    return { exito: false, mensaje: "No encontrado" };
  } catch(e) { return { exito: false, mensaje: e.toString() }; }
}

/**
 * Avanza el estado de la constancia (Generar o Finalizar) y notifica al alumno.
 */
function avanzarConstancia(id, accion, adminData = null) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Constancias");
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    let rowData = null;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(id).trim()) {
        rowIndex = i + 1;
        rowData = data[i];
        break;
      }
    }
    
    if (rowIndex === -1) return { exito: false, mensaje: "Solicitud no encontrada" };

    if (accion === 'Generar') {
      sheet.getRange(rowIndex, 6).setValue('En Proceso'); 
      if (adminData) {
        sheet.getRange(rowIndex, 8).setValue(JSON.stringify(adminData)); 
      }
      return { exito: true, nuevoEstatus: 'En Proceso' };
    } 
    else if (accion === 'Finalizar') {
      sheet.getRange(rowIndex, 6).setValue('Terminada'); 
      const email = rowData[3];
      const studentName = rowData[2];
      
      if (email) {
        MailApp.sendEmail({
          to: email,
          subject: `Tu Constancia de Estudios está Lista - #${id}`,
          htmlBody: `
            <div style="font-family: Arial, sans-serif; padding: 20px; color: #333; max-width: 600px; border: 1px solid #ddd; border-radius: 8px;">
              <h2 style="color: #059669;">Constancia Terminada y Firmada</h2>
              <p>Estimado/a <strong>${studentName}</strong>,</p>
              <p>Te informamos formalmente que tu solicitud de Constancia de Estudios con el folio <strong>#${id}</strong> ha sido procesada exitosamente.</p>
              <p style="background-color: #ecfdf5; padding: 15px; border-left: 4px solid #059669; font-weight: bold; color: #064e3b;">
                Por favor, pasa a la Coordinación de Ingeniería en Computación (Módulo M202) con tu identificación oficial para recoger tu documento físico.
              </p>
              <p style="font-size: 12px; color: #666; margin-top: 20px;">Atentamente,<br/>Coordinación INCO - CUCEI</p>
            </div>
          `
        });
      }
      return { exito: true, nuevoEstatus: 'Terminada' };
    }
  } catch(e) { 
    return { exito: false, mensaje: "Error: " + e.toString() }; 
  }
}
