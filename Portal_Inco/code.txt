/**
 * 1. CREAR UN NUEVO TICKET (Vista Alumno)
 */
function createTicket(ticketData, userEmail) {
  const lock = LockService.getScriptLock();
  try {
    // Mandato: Sanitización estricta de inputs
    const safeSubject = sanitizeInput(ticketData.asunto);
    const safeDetails = sanitizeInput(ticketData.detalles);
    const safeCode = sanitizeInput(ticketData.codigo);
    const safeName = sanitizeInput(ticketData.nombre);
    const safeEmail = sanitizeInput(userEmail);

    if (!lock.tryLock(5000)) {
      return { success: false, message: "El servidor está procesando otras solicitudes, intenta en 5 segundos." };
    }

    const ssId = PropertiesService.getScriptProperties().getProperty('DB_SHEET_ID');
    const sheet = SpreadsheetApp.openById(ssId).getSheetByName(TABLES.TICKETS);
    
    // Generar Folio único. Ej: INC-1698425000
    const folioId = `INC-${Math.floor(Date.now() / 1000)}`;
    const fechaActual = Utilities.formatDate(new Date(), "America/Mexico_City", "dd MMM yyyy HH:mm");

    // Esquema: id_ticket, codigo_estudiante, nombre_estudiante, asunto, detalles, estatus, respuesta, fecha_creacion, correo_estudiante (añadido logico)
    const newRow = [
      folioId, 
      safeCode, 
      safeName, 
      safeSubject, 
      safeDetails, 
      'Pendiente', 
      '', 
      fechaActual, 
      safeEmail // Columna oculta extra para enviar la respuesta más fácil
    ];

    sheet.appendRow(newRow);
    SpreadsheetApp.flush();
    
    // Mandato: Auditoría
    logSystemEvent('INFO', safeCode, 'Ticket Creado', `Se aperturó el ticket folio ${folioId}`);

    // Mandato: Envío de Correo HTML - Recepción
    enviarCorreoTicket(safeEmail, safeName, 'creacion', {
      folio: folioId,
      asunto: safeSubject,
      detalles: safeDetails,
      fecha: fechaActual
    });

    return {
      success: true,
      ticket: {
        id_ticket: folioId,
        asunto: safeSubject,
        detalles: safeDetails,
        estatus: 'Pendiente',
        fecha_creacion: fechaActual,
        respuesta_coordinacion: ''
      }
    };

  } catch (error) {
    logSystemEvent('ERROR', ticketData.codigo || 'SISTEMA', 'Error en Creación Ticket', error.toString());
    return { success: false, message: "Error interno al crear el ticket." };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 2. OBTENER TICKETS DEL ALUMNO (Vista Alumno)
 * Prevención IDOR: Solo devuelve los tickets que coinciden con el código que solicita.
 */
function getStudentTickets(studentCode) {
  try {
    const ssId = PropertiesService.getScriptProperties().getProperty('DB_SHEET_ID');
    const sheet = SpreadsheetApp.openById(ssId).getSheetByName(TABLES.TICKETS);
    
    // Regla de Optimización: getValues en lote
    const data = sheet.getDataRange().getValues();
    const result = [];

    // Omitimos fila 0 (headers)
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      // Mandato Seguridad: Aislamiento (IDOR). Solo si el código de la BD (row 1) coincide.
      if (String(row[1]) === String(studentCode)) {
        result.push({
          id_ticket: row[0],
          asunto: row[3],
          detalles: row[4],
          estatus: row[5],
          respuesta_coordinacion: row[6],
          fecha_creacion: row[7]
        });
      }
    }
    
    // Devolvemos el array invertido para mostrar los más recientes primero
    return result.reverse();

  } catch (error) {
    logSystemEvent('ERROR', studentCode, 'Error Lectura Tickets', error.toString());
    throw new Error("No se pudieron cargar los tickets.");
  }
}

/**
 * 3. OBTENER TODOS LOS TICKETS (Vista Coordinador)
 * Usa CacheService porque los tickets históricos no cambian cada segundo.
 */
function getAllTickets() {
  try {
    const cache = CacheService.getScriptCache();
    const cachedData = cache.get('ADMIN_ALL_TICKETS');
    
    if (cachedData) {
      return JSON.parse(cachedData);
    }

    const ssId = PropertiesService.getScriptProperties().getProperty('DB_SHEET_ID');
    const sheet = SpreadsheetApp.openById(ssId).getSheetByName(TABLES.TICKETS);
    
    const data = sheet.getDataRange().getValues();
    const result = [];

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      result.push({
        id_ticket: row[0],
        codigo_estudiante: row[1],
        nombre_estudiante: row[2],
        asunto: row[3],
        detalles: row[4],
        estatus: row[5],
        respuesta_coordinacion: row[6],
        fecha_creacion: row[7],
        correo_estudiante: row[8] // Necesario para responder
      });
    }
    
    const finalData = result.reverse();
    cache.put('ADMIN_ALL_TICKETS', JSON.stringify(finalData), 60); // Cache por 60 segundos
    return finalData;

  } catch (error) {
    logSystemEvent('ERROR', 'ADMIN', 'Error Lectura Global Tickets', error.toString());
    throw new Error("No se pudieron cargar los tickets globales.");
  }
}

/**
 * 4. RESPONDER A UN TICKET (Vista Coordinador)
 */
function replyToTicket(ticketId, replyText, studentEmail, studentName) {
  const lock = LockService.getScriptLock();
  try {
    const safeReply = sanitizeInput(replyText);
    
    if (!lock.tryLock(5000)) return { success: false, message: "Sistema ocupado." };

    const ssId = PropertiesService.getScriptProperties().getProperty('DB_SHEET_ID');
    const sheet = SpreadsheetApp.openById(ssId).getSheetByName(TABLES.TICKETS);
    const data = sheet.getDataRange().getValues();
    
    let rowIndex = -1;
    let ticketSubject = '';
    let ticketDetails = '';

    // Buscar en memoria (Optimización)
    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === ticketId) {
        rowIndex = i + 1; // +1 porque el array es index 0 y sheets es index 1
        ticketSubject = data[i][3];
        ticketDetails = data[i][4];
        break;
      }
    }

    if (rowIndex === -1) {
      return { success: false, message: "Ticket no encontrado en la base de datos." };
    }

    // Actualizar columnas específicas en Sheets (Fila encontrada, Columna 6 Estatus, Col 7 Respuesta, Col 9 Fecha_Update)
    const fechaUpdate = Utilities.formatDate(new Date(), "America/Mexico_City", "dd MMM yyyy HH:mm");
    
    // Escritura puntual
    sheet.getRange(rowIndex, 6, 1, 4).setValues([['Atendido', safeReply, fechaUpdate, studentEmail]]);
    SpreadsheetApp.flush();

    // Invalidar caché global para que el admin vea la actualización inmediata en su F5
    CacheService.getScriptCache().remove('ADMIN_ALL_TICKETS');

    // Mandato: Auditoría
    logSystemEvent('INFO', 'ADMIN_COORDINACION', 'Ticket Atendido', `Se respondió al ticket ${ticketId}`);

    // Mandato: Envío de Correo HTML - Respuesta
    if (studentEmail) {
      enviarCorreoTicket(studentEmail, studentName, 'respuesta', {
        folio: ticketId,
        asunto: ticketSubject,
        detalles: ticketDetails,
        respuesta: safeReply
      });
    }

    return { success: true };

  } catch (error) {
    logSystemEvent('ERROR', 'ADMIN_COORDINACION', 'Error Cerrando Ticket', error.toString());
    return { success: false, message: "Error interno al procesar la respuesta." };
  } finally {
    lock.releaseLock();
  }
}
